```shell
# 使用 ssh-keygen 生成一个 ssh key
# 目前使用密码上不如使用 key 安全的，因为 key 的长度通常是 2048 或 4096 位，远超普通的密码。这里建议按照 GitHub 官方网站上的教程生成新 SSH 密钥并添加到 ssh-agent，这样不仅可以用于登录 Linux 机器，也可以用于 git。

# 使用 ssh-keygen 生成了密钥以后，你会得到两个文件：

# ~/.ssh/id_rsa
# ~/.ssh/id_rsa.pub
# 其中 id_rsa 是私钥，id_rsa.pub 是公钥。不要把私钥发给任何人。
# 首先使用密码登录刚才的机器, 然后创建 ~/.ssh 目录，并且创建 ~/.ssh/authorized_keys 文件，配置好相应的权限：
mkdir -p ~/.ssh
touch ~/.ssh/authorized_keys
chmod 700 ~/.ssh
chmod 600 ~/.ssh/authorized_keys
nano ~/.ssh/authorized_keys

# 添加你的公钥，如 ssh-rsa AAAA...
# 把刚才生成的公钥 id_rsa.pub 添加到 ~/.ssh/authorized_keys 里，如果文件已经有内容，就添加在最后一行。这样以后每次使用 ssh 登录机器都不需要再输入密码了。
# 禁止密码登录
# 由于密码登录存在不安全因素，比如暴露在公网的 IP 会被扫描，而 key 是相对安全的，所以我们可以禁止不安全的密码登录：
sudo nano /etc/ssh/sshd_config

PasswordAuthentication no # 添加在最后一行
```

### 服务器代理挂载

为了方便解决问题，统一使用 `Clash Verge`

```shell
https://www.clashverge.dev/ # 自行下载
# 比较便宜的机场推荐，非广告
https://www.mojie.co
https://hk.freecat.cloud/
# 购买后在Clash Verge中添加订阅即可

# 服务器挂载代理，在设置中开启局域网（不要和公网ip混淆来问这种愚蠢的问题）连接，查看自己的ip并记下端口，在服务器端的环境变量中设置如下脚本代码：
# 定义代理地址（根据实际端口修改）
proxy_on() {
    # 默认值（可自行调整）
    local ip="${1:-xxx,xxx,xxx,xxx}" # 自行改为自己的ip地址
    local port="${2:-xxx}" # 自行改为自己的端口

    # 构造代理地址
    export proxy="http://$ip:$port"
    export http_proxy="$proxy"
    export https_proxy="$proxy"
    export ftp_proxy="$proxy"
    echo "代理已启用：$proxy"
}

# 禁用代理
proxy_off() {
    unset proxy http_proxy https_proxy ftp_proxy
    echo "代理已禁用"
}

# 检查代理状态
proxy_status() {
    if [ -n "$http_proxy" ]; then
        echo "当前代理状态：启用 ($http_proxy)"
    else
        echo "当前代理状态：未启用"
    fi
}
# 使用方法
proxy_on # 启用默认代理
proxy_on your_ip your_port # 启用代理，并指定ip和端口
proxy_off # 禁用代理
proxy_status # 查看代理状态

# 或者采用以下手动方式：
export proxy="http://xxx.xxx.xxx.xxx:xxxx" # 修改为你的代理地址和端口
export http_proxy=$proxy
export https_proxy=$proxy
export ftp_proxy=$proxy
```
